import { access, mkdir, mkdtemp, rename, rm } from "node:fs/promises";
import { homedir } from "node:os";
import { spawnSync } from "node:child_process";
import { join } from "node:path";
import process from "node:process";
import payload from "./runtime.tar.gz" with { type: "file" };

// The build replaces this literal with the archive's content hash.
const buildId = "FUNGI_ARCHIVE_HASH";
const base = join(process.env.FUNGI_INSTALL ?? join(homedir(), ".fungi"), "runtime");
const runtime = join(base, buildId);
try {
  await access(join(runtime, "package.json"));
} catch {
  await mkdir(base, { recursive: true, mode: 0o700 });
  const temporary = await mkdtemp(join(base, ".extract-"));
  try {
    await new Bun.Archive(await Bun.file(payload).arrayBuffer()).extract(temporary);
    try { await rename(temporary, runtime); }
    catch (cause) {
      // Another launch may have completed the same immutable extraction.
      if (!(cause instanceof Error && "code" in cause && (cause.code === "ENOTEMPTY" || cause.code === "EEXIST"))) throw cause;
    }
  } finally { await rm(temporary, { recursive: true, force: true }); }
}
const cli = join(runtime, "dist/cli.js");
process.env.FUNGI_BUNDLED_CLI = cli;
process.env.FUNGI_BUNDLED_BUILD_ID = buildId;
process.env.FUNGI_LAUNCHER = process.execPath;
process.env.FUNGI_LAUNCHER_NODE = process.execPath;
process.env.FUNGI_NODE = join(runtime, "native/bin/node");
// Existing service/restart callers pass the CLI/launcher before its arguments.
const args = process.argv.slice(2);
if (args[0] === cli || args[0] === process.execPath) args.shift();
const child = spawnSync(join(runtime, "native/bin/bun"), [cli, ...args], {
  stdio: "inherit",
  env: process.env,
});
if (child.error) {
  process.stderr.write("fungi: Could not start the embedded runtime. Reinstall fungi.\n");
  process.exitCode = 1;
} else if (child.signal) {
  process.kill(process.pid, child.signal);
} else {
  process.exitCode = child.status ?? 1;
}
